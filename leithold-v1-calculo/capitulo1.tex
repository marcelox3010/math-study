\section*{Exercícios Resolvidos}

\begin{exercicio}
\( 5x + 2 > x - 6 \)
\end{exercicio}

\textbf{Solução:}

\begin{align}
5x - x &> -6 - 2 \\
4x &> -8 \\
x &> \frac{-8}{4} \\
x &> -2
\end{align}

\hrulefill{}

\begin{exercicio}
 $3-x<5+3x$.
\end{exercicio}

\textbf{Solução:}
\setcounter{equation}{0}
\begin{align}
-x-3x&<5-3\\
-4x&<2\\
-x&<\frac{2}{4}\\
-x&<\frac{1}{2} \quad \text{(multiplicando por -1)}\\
x&>-\frac{1}{2}
\end{align}

\hrulefill{}

\begin{exercicio}
 \( \frac{2}{3}x - \frac{1}{2} \leq 0 \).
\end{exercicio}

\textbf{Solução:}
\setcounter{equation}{0}
\begin{align}
\frac{2}{3}x \leq \frac{1}{2} \\
\frac{2}{3}x \leq \frac{1}{4} \\
x \leq \frac{3}{2} \cdot \frac{1}{2} \\
x \leq \frac{3}{4} 
\end{align}
