\section*{Exercícios Resolvidos}

\begin{exercicio}
\( 5x + 2 > x - 6 \)
\end{exercicio}

\textbf{Solução:}

\begin{align}
5x + 2 &> x - 6 \quad \textcolor{blue}{(-x)}\\
4x + 2 &> - 6 \quad \textcolor{blue}{(-2)} \\
4x &> -8 \quad \textcolor{blue}{(\div\ 4)}\\
x &> -2\\
\end{align}

\[
x \in (-2, +\infty)
\]

\hrulefill{}

\begin{exercicio}
 $3-x<5+3x$.
\end{exercicio}

\textbf{Solução:}
\setcounter{equation}{0}
\begin{align}
-x-3x&<5-3\\
-4x&<2\\
-x&<\frac{2}{4}\\
-x&<\frac{1}{2} \quad \textcolor{blue}{(\cdot -1)}\\
x&>-\frac{1}{2}
\end{align}

\hrulefill{}

\begin{exercicio}
 \( \frac{2x}{3} - \frac{1}{2} \leq 0 \).
\end{exercicio}

\textbf{Solução:}
\setcounter{equation}{0}
\begin{align}
\frac{2x}{3} \leq \frac{1}{2} \\
\frac{2x}{3} \leq \frac{1}{4} \\
x \leq \frac{3}{2} \cdot \frac{1}{2} \\
x \leq \frac{3}{4} 
\end{align}

\hrulefill{}

\begin{exercicio}
\( 3-2x \geq 9+4x \)
\end{exercicio}

\textbf{Solução:}
\setcounter{equation}{0}
\begin{align}
-2x-4x \geq 9-3 \\
-6x \geq 6 \\
-x \geq 1 \quad \textcolor{blue}{(\cdot -1)}\\
x \leq -1
\end{align}

\hrulefill{}

\begin{exercicio}
\(13 \geq 2x-3 \geq 5\)
\end{exercicio}

\textbf{Solução:}
\setcounter{equation}{0}
\begin{align}
13 \geq 2x-3 \geq 5 \quad \textcolor{blue}{(+ 3)}\\
16 \geq 2x \geq 8 \quad \textcolor{blue}{(\div\ 2)}\\
8 \geq x \geq 4 \\
x \in [4,8], x \ge 4,\ x \le 8
\end{align}

\hrulefill{}
