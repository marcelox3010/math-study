\begin{exercicio}
Calcule a derivada de \( f(x) = x^2 + 3x \).
\end{exercicio}

\textbf{Solução:}

\[
f'(x) = 2x + 3
\]

\hrulefill

\begin{exercicio}
Resolva a equação \( x^2 - 5x + 6 = 0 \).
\end{exercicio}

\textbf{Solução:}

\[
x^2 - 5x + 6 = (x-2)(x-3)
\]

\[
x = 2 \quad \text{ou} \quad x = 3
\]
